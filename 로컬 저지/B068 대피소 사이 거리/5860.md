### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
n, m = int(data[0]), int(data[1])
pos = 2
adj = [[] for _ in range(n + 1)]
for _ in range(n - 1):
    a, b, w = int(data[pos]), int(data[pos + 1]), int(data[pos + 2])
    pos += 3
    adj[a].append((b, w))
    adj[b].append((a, w))
par = [0] * (n + 1)
dep = [0] * (n + 1)
dist = [0] * (n + 1)
seen = [False] * (n + 1)
seen[1] = True
stack = [1]
while stack:
    u = stack.pop()
    for v, w in adj[u]:
        if not seen[v]:
            seen[v] = True
            par[v] = u
            dep[v] = dep[u] + 1
            dist[v] = dist[u] + w
            stack.append(v)
out = []
for _ in range(m):
    a, b = int(data[pos]), int(data[pos + 1])
    pos += 2
    x, y = a, b
    while dep[x] > dep[y]:
        x = par[x]
    while dep[y] > dep[x]:
        y = par[y]
    while x != y:
        x = par[x]
        y = par[y]
    out.append(str(dist[a] + dist[b] - 2 * dist[x]))
print("\n".join(out))
```
