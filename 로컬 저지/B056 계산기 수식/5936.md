### 파이썬 코드

```python
import sys
from itertools import product

data = sys.stdin.read().split()
t = int(data[0])
out = []
for i in range(1, t + 1):
    n = int(data[i])
    for ops in product(" +-", repeat=n - 1):
        expr = "1"
        for k in range(n - 1):
            expr += ops[k] + str(k + 2)
        if eval(expr.replace(" ", "")) == 0:
            out.append(expr)
    out.append("")
sys.stdout.write("\n".join(out) + "\n")
```
