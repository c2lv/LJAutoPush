### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
n, L, R = int(data[0]), int(data[1]), int(data[2])
g = [int(v) for v in data[3:3 + n * n]]
nb = []
for p in range(n * n):
    r, c = divmod(p, n)
    lst = []
    if r > 0:
        lst.append(p - n)
    if r < n - 1:
        lst.append(p + n)
    if c > 0:
        lst.append(p - 1)
    if c < n - 1:
        lst.append(p + 1)
    nb.append(lst)
cand = list(range(n * n))
rounds = 0
while True:
    seen = bytearray(n * n)
    groups = []
    for s in cand:
        if seen[s]:
            continue
        seen[s] = 1
        comp = [s]
        stack = [s]
        while stack:
            p = stack.pop()
            gp = g[p]
            for q in nb[p]:
                if not seen[q]:
                    d = gp - g[q]
                    if d < 0:
                        d = -d
                    if L <= d <= R:
                        seen[q] = 1
                        comp.append(q)
                        stack.append(q)
        if len(comp) > 1:
            groups.append(comp)
    if not groups:
        break
    rounds += 1
    nxt = []
    for comp in groups:
        avg = sum(g[p] for p in comp) // len(comp)
        for p in comp:
            g[p] = avg
        nxt.extend(comp)
    cand = nxt
print(rounds)
```
