### 파이썬 코드

```python
import sys

data = sys.stdin.buffer.read().split()
n = int(data[0])
m = int(data[1])
INF = float("inf")
d = [[INF] * n for _ in range(n)]
for i in range(n):
    d[i][i] = 0
pos = 2
for _ in range(m):
    a, b, c = int(data[pos]) - 1, int(data[pos + 1]) - 1, int(data[pos + 2])
    pos += 3
    if c < d[a][b]:
        d[a][b] = c
for k in range(n):
    dk = d[k]
    for i in range(n):
        dik = d[i][k]
        if dik == INF:
            continue
        di = d[i]
        for j in range(n):
            v = dik + dk[j]
            if v < di[j]:
                di[j] = v
out = []
for i in range(n):
    out.append("".join(("0" if v == INF else str(v)) + " " for v in d[i]))
sys.stdout.write("\n".join(out) + "\n")
```
