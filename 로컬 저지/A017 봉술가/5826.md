```python
def solution(N, board):
    W = N + 2
    dist = [0] * (W * W)
    for i in range(N):
        base = (i + 1) * W + 1
        row = board[i]
        for j in range(N):
            dist[base + j] = 0 if row[j] == "G" else -1
    cur = [p for p in range(W * W) if dist[p] == 0 and W <= p < W * (W - 1) and 0 < p % W < W - 1]
    d = 0
    dirs = (1, -1, W, -W)
    while cur:
        d += 1
        nxt = []
        for p in cur:
            for dd in dirs:
                q = p + dd
                if dist[q] == -1:
                    dist[q] = d
                    nxt.append(q)
        cur = nxt
    best = -1
    ans = [0, 0]
    for i in range(N):
        base = (i + 1) * W + 1
        row = board[i]
        for j in range(N):
            if row[j] != "G" and dist[base + j] > best:
                best = dist[base + j]
                ans = [i, j]
    return ans
```
