### 파이썬 코드

```python
import sys

lines = sys.stdin.read().split("\n")
n, k, m = map(int, lines[0].split()[:3])
books = []
li = 1
while len(books) < m and li < len(lines):
    parts = lines[li].split()
    li += 1
    if not parts and len(books) < m and li < len(lines) and all(not l.strip() for l in lines[li:]):
        books.append([])
        continue
    books.append([int(v) for v in parts])
lang_books = [[] for _ in range(n + 1)]
for b, langs in enumerate(books):
    for l in langs:
        lang_books[l].append(b)
if n == 1:
    print(1)
else:
    dist = [0] * (n + 1)
    dist[1] = 1
    used = [False] * len(books)
    cur = [1]
    ans = -1
    while cur and ans == -1:
        nxt = []
        for l in cur:
            for b in lang_books[l]:
                if used[b]:
                    continue
                used[b] = True
                for l2 in books[b]:
                    if dist[l2] == 0:
                        dist[l2] = dist[l] + 1
                        if l2 == n:
                            ans = dist[l2]
                        nxt.append(l2)
        cur = nxt
    print(ans)
```
