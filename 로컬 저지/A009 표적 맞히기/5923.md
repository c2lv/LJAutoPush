```python
def solution(N, K, board):
    vec = {"U": (-1, 0), "R": (0, 1), "D": (1, 0), "L": (0, -1)}
    cw = {"U": "R", "R": "D", "D": "L", "L": "U"}
    shots = []
    for c in range(N):
        shots.append((0, c, 1, 0))
        shots.append((N - 1, c, -1, 0))
    for r in range(N):
        shots.append((r, N - 1, 0, -1))
        shots.append((r, 0, 0, 1))

    def fire(g, s):
        r, c, dr, dc = s
        while 0 <= r < N and 0 <= c < N:
            ch = g[r][c]
            if ch == "X":
                return 0, None
            if ch in vec:
                tr, tc = vec[ch]
                if tr == -dr and tc == -dc:
                    return 3, (r, c, "B")
                if tr == dr and tc == dc:
                    return 1, None
                return 2, (r, c, cw[ch])
            r += dr
            c += dc
        return 0, None

    best = 0

    def dfs(g, k, score):
        nonlocal best
        if k == 3:
            if score > best:
                best = score
            return
        for s in shots:
            pts, ch = fire(g, s)
            if ch is None:
                dfs(g, k + 1, score + pts)
            else:
                r, c, nv = ch
                old = g[r][c]
                g[r][c] = nv
                dfs(g, k + 1, score + pts)
                g[r][c] = old

    dfs([list(row) for row in board], 0, 0)
    return best
```
