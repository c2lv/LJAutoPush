```python
from bisect import bisect_left, bisect_right

def solution(N, K, arr, queries):
    a = sorted(arr)
    res = []
    for x, y in queries:
        c = bisect_right(a, y) - bisect_left(a, x)
        res.append(c if c > 0 else 0)
    return res
```
