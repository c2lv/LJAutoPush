### 파이썬 코드

```python
import sys

data = sys.stdin.buffer.read().split()
t = int(data[0])
pos = 1
out = []
for _ in range(t):
    n = int(data[pos])
    pos += 1
    col = [0] * (n + 1)
    for i in range(n):
        col[int(data[pos])] = int(data[pos + 1])
        pos += 2
    best = n + 1
    cnt = 0
    for r in range(1, n + 1):
        c = col[r]
        if c < best:
            best = c
            cnt += 1
    out.append(str(cnt))
sys.stdout.write("\n".join(out) + "\n")
```
