import sys

data = sys.stdin.read().split()
t = int(data[0])
z = [1, 0]
o = [0, 1]
for i in range(2, 41):
    z.append(z[-1] + z[-2])
    o.append(o[-1] + o[-2])
out = []
for i in range(1, t + 1):
    n = int(data[i])
    out.append("%d %d" % (z[n], o[n]))
print("\n".join(out))
