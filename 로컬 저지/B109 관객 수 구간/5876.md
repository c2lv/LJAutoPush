### 파이썬 코드

```python
import sys

data = sys.stdin.buffer.read().split()
n, x = int(data[0]), int(data[1])
a = list(map(int, data[2:2 + n]))
s = sum(a[:x])
best = s
cnt = 1
for i in range(x, n):
    s += a[i] - a[i - x]
    if s > best:
        best = s
        cnt = 1
    elif s == best:
        cnt += 1
if best == 0:
    print("SAD")
else:
    print(best)
    print(cnt)
```
