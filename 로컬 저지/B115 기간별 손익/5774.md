### 파이썬 코드

```python
import sys

input = sys.stdin.read
data = input().split()

N = int(data[0])
profits = list(map(int, data[1:N+1]))
M = int(data[N+1])

prefix_sum = [0] * (N + 1)
for i in range(N):
    prefix_sum[i+1] = prefix_sum[i] + profits[i]

result = []
for i in range(N+2, N+2+2*M, 2):
    start = int(data[i]) - 1
    end = int(data[i+1])
    result.append(str(prefix_sum[end] - prefix_sum[start]))

print('\n'.join(result))
```
