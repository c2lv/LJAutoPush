### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
n = int(data[0])
w = [[int(data[1 + i * n + j]) for j in range(n)] for i in range(n)]
INF = float("inf")
full = 1 << n
dp = [[INF] * n for _ in range(full)]
dp[1][0] = 0
for mask in range(1, full, 2):
    row = dp[mask]
    for last in range(n):
        v = row[last]
        if v == INF:
            continue
        wl = w[last]
        for nx in range(n):
            if (mask >> nx) & 1 or wl[nx] == 0:
                continue
            nm = mask | (1 << nx)
            nv = v + wl[nx]
            if nv < dp[nm][nx]:
                dp[nm][nx] = nv
best = INF
for last in range(1, n):
    if w[last][0] and dp[full - 1][last] + w[last][0] < best:
        best = dp[full - 1][last] + w[last][0]
print(best)
```
