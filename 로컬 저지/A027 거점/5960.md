```python
import heapq
from itertools import permutations

def solution(N, M, edges, points):
    adj = [[] for _ in range(N + 1)]
    for a, b, w in edges:
        adj[a].append((b, w))
        adj[b].append((a, w))
    INF = float("inf")

    def dijkstra(src):
        dist = [INF] * (N + 1)
        dist[src] = 0
        pq = [(0, src)]
        while pq:
            d, u = heapq.heappop(pq)
            if d > dist[u]:
                continue
            for v, w in adj[u]:
                nd = d + w
                if nd < dist[v]:
                    dist[v] = nd
                    heapq.heappush(pq, (nd, v))
        return dist

    srcs = [1] + list(points)
    D = {s: dijkstra(s) for s in srcs}
    best = INF
    for p in permutations(points):
        total = D[1][p[0]] + D[p[0]][p[1]] + D[p[1]][p[2]] + D[p[2]][N]
        if total < best:
            best = total
    return -1 if best == INF else best
```
