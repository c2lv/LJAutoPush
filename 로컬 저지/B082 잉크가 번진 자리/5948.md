### 파이썬 코드

```python
import sys
from collections import deque

data = sys.stdin.read().split()
n, k = int(data[0]), int(data[1])
g = [[int(data[2 + i * n + j]) for j in range(n)] for i in range(n)]
p = 2 + n * n
s, x, y = int(data[p]), int(data[p + 1]) - 1, int(data[p + 2]) - 1
cells = sorted((g[i][j], i, j) for i in range(n) for j in range(n) if g[i][j])
q = deque((v, i, j, 0) for v, i, j in cells)
while q:
    v, i, j, t = q.popleft()
    if t == s:
        break
    for ni, nj in ((i + 1, j), (i - 1, j), (i, j + 1), (i, j - 1)):
        if 0 <= ni < n and 0 <= nj < n and g[ni][nj] == 0:
            g[ni][nj] = v
            q.append((v, ni, nj, t + 1))
print(g[x][y])
```
