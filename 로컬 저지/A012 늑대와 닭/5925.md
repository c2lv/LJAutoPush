def solution(N, K, arr):
    wolves = []
    chick = []
    for i in range(N):
        for j in range(N):
            if arr[i][j] == "W":
                wolves.append((i, j))
            elif arr[i][j] == "C":
                chick.append((i, j))
    INF = float("inf")
    D = []
    for sx, sy in wolves:
        dist = [[-1] * N for _ in range(N)]
        dist[sx][sy] = 0
        cur = [(sx, sy)]
        d = 0
        while cur:
            d += 1
            nxt = []
            for x, y in cur:
                for nx, ny in ((x + 1, y), (x - 1, y), (x, y + 1), (x, y - 1)):
                    if 0 <= nx < N and 0 <= ny < N and dist[nx][ny] == -1 and arr[nx][ny] != "X":
                        dist[nx][ny] = d
                        nxt.append((nx, ny))
            cur = nxt
        D.append([dist[cx][cy] if dist[cx][cy] != -1 else INF for cx, cy in chick])
    k = len(wolves)
    m = len(chick)
    dp = [INF] * (1 << m)
    dp[0] = 0
    for mask in range(1 << m):
        if dp[mask] == INF:
            continue
        i = bin(mask).count("1")
        if i >= k:
            continue
        for c in range(m):
            if not (mask >> c) & 1:
                v = max(dp[mask], D[i][c])
                nm = mask | (1 << c)
                if v < dp[nm]:
                    dp[nm] = v
    best = min(dp[mask] for mask in range(1 << m) if bin(mask).count("1") == k)
    return 0 if best == INF else best
