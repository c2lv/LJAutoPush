### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
t = int(data[0])
pos = 1
out = []
for _ in range(t):
    n = int(data[pos])
    vals = [int(x) for x in data[pos + 1:pos + 1 + n]]
    pos += 1 + n
    s = sum(vals)
    cnt = sum(1 for v in vals if v * n > s)
    q = (cnt * 200000 + n) // (2 * n)
    out.append("%d.%03d%%" % (q // 1000, q % 1000))
print("\n".join(out))
```
