### 파이썬 코드

```python
import sys

data = sys.stdin.buffer.read().split()
t = int(data[0])
pos = 1
out = []
for _ in range(t):
    v, m = int(data[pos]), int(data[pos + 1])
    pos += 2
    adj = [[] for _ in range(v + 1)]
    for i in range(m):
        a, b = int(data[pos]), int(data[pos + 1])
        pos += 2
        adj[a].append(b)
        adj[b].append(a)
    color = [0] * (v + 1)
    ok = True
    for s in range(1, v + 1):
        if color[s] or not ok:
            continue
        color[s] = 1
        stack = [s]
        while stack and ok:
            u = stack.pop()
            cu = color[u]
            for w in adj[u]:
                if color[w] == 0:
                    color[w] = -cu
                    stack.append(w)
                elif color[w] == cu:
                    ok = False
                    break
    out.append("YES" if ok else "NO")
print("\n".join(out))
```
