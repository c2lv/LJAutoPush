### 파이썬 코드

```python
import sys

n = int(sys.stdin.read().split()[0])
full = (1 << n) - 1


def solve(cols, ld, rd):
    if cols == full:
        return 1
    total = 0
    avail = full & ~(cols | ld | rd)
    while avail:
        bit = avail & -avail
        avail ^= bit
        total += solve(cols | bit, ((ld | bit) << 1) & full, (rd | bit) >> 1)
    return total


if n == 1:
    print(1)
else:
    res = 0
    for c in range(n // 2):
        bit = 1 << c
        res += solve(bit, (bit << 1) & full, bit >> 1)
    res *= 2
    if n % 2 == 1:
        bit = 1 << (n // 2)
        res += solve(bit, (bit << 1) & full, bit >> 1)
    print(res)
```
