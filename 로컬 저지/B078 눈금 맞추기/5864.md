### 파이썬 코드

```python
import sys
from collections import deque

n, k = map(int, sys.stdin.read().split()[:2])
L = 100000
dist = [-1] * (L + 1)
dist[n] = 0
q = deque([n])
while q:
    x = q.popleft()
    if x == k:
        break
    d = dist[x] + 1
    for y in (x - 1, x + 1, 2 * x):
        if 0 <= y <= L and dist[y] == -1:
            dist[y] = d
            q.append(y)
print(dist[k])
```
