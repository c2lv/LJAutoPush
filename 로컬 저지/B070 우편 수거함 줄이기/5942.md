### 파이썬 코드

```python
import sys
from itertools import combinations

data = sys.stdin.read().split()
n, m = int(data[0]), int(data[1])
houses = []
boxes = []
for i in range(n):
    for j in range(n):
        v = data[2 + i * n + j]
        if v == "1":
            houses.append((i, j))
        elif v == "2":
            boxes.append((i, j))
D = [[abs(hx - bx) + abs(hy - by) for (bx, by) in boxes] for (hx, hy) in houses]
best = None
for comb in combinations(range(len(boxes)), m):
    s = 0
    for row in D:
        s += min(row[b] for b in comb)
        if best is not None and s >= best:
            break
    if best is None or s < best:
        best = s
print(best)
```
