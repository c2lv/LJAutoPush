import sys

data = sys.stdin.read().split()
n = int(data[0])
a = b = c = 0
for i in range(n):
    x, y, z = int(data[1 + 3 * i]), int(data[2 + 3 * i]), int(data[3 + 3 * i])
    a, b, c = x + min(b, c), y + min(a, c), z + min(a, b)
print(min(a, b, c))
