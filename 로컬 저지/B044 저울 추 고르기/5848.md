### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
t = int(data[0])
fib = [1, 2]
while fib[-1] <= 2000000000:
    fib.append(fib[-1] + fib[-2])
out = []
for i in range(1, t + 1):
    n = int(data[i])
    used = []
    for f in reversed(fib):
        if f <= n:
            used.append(f)
            n -= f
    used.reverse()
    out.append("".join("%d " % u for u in used))
sys.stdout.write("\n".join(out) + "\n")
```
