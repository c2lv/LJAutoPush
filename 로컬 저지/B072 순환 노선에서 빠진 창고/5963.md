### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
t = int(data[0])
pos = 1
out = []
for _ in range(t):
    n = int(data[pos])
    pos += 1
    to = [0] + [int(v) for v in data[pos:pos + n]]
    pos += n
    indeg = [0] * (n + 1)
    for i in range(1, n + 1):
        indeg[to[i]] += 1
    stack = [i for i in range(1, n + 1) if indeg[i] == 0]
    removed = 0
    while stack:
        u = stack.pop()
        removed += 1
        v = to[u]
        indeg[v] -= 1
        if indeg[v] == 0:
            stack.append(v)
    out.append(str(removed))
print("\n".join(out))
```
