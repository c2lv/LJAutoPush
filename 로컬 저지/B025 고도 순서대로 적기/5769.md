```python
import sys

# 입력 받기
N = int(sys.stdin.readline())
heights = [int(sys.stdin.readline()) for _ in range(N)]

# 정렬
heights.sort()

# 출력
for height in heights:
    print(height)
print()  # 빈 줄 추가
```
