### 파이썬 코드

```python
from itertools import product

def solution(N, K, arr):
    best = None
    for assign in product(range(K), repeat=N):
        cnt = [0] * K
        s = [0] * K
        for i in range(N):
            g = assign[i]
            cnt[g] += 1
            s[g] += arr[i]
        if 0 in cnt:
            continue
        powers = [cnt[g] * s[g] for g in range(K)]
        diff = max(powers) - min(powers)
        if best is None or diff < best:
            best = diff
    return best
```
