### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
n, m = int(data[0]), int(data[1])
pos = 2
ps = [[0] * (m + 1) for _ in range(n + 1)]
for i in range(1, n + 1):
    row = ps[i]
    up = ps[i - 1]
    run = 0
    for j in range(1, m + 1):
        run += int(data[pos])
        pos += 1
        row[j] = up[j] + run
k = int(data[pos])
pos += 1
out = []
for _ in range(k):
    i, j, x, y = int(data[pos]), int(data[pos + 1]), int(data[pos + 2]), int(data[pos + 3])
    pos += 4
    out.append(str(ps[x][y] - ps[i - 1][y] - ps[x][j - 1] + ps[i - 1][j - 1]))
print("\n".join(out))
```
