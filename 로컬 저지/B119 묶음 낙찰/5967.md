### 파이썬 코드

```python
import sys

data = sys.stdin.buffer.read().split()
n = int(data[0])
shop = data[1:2 * n + 1:2]
val = list(map(int, data[2:2 * n + 2:2]))
order = sorted(range(n), key=val.__getitem__)
ans = [0] * n
shop_sum = {}
total = 0
i = 0
while i < n:
    j = i
    v = val[order[i]]
    while j < n and val[order[j]] == v:
        j += 1
    for k in range(i, j):
        idx = order[k]
        ans[idx] = total - shop_sum.get(shop[idx], 0)
    for k in range(i, j):
        idx = order[k]
        s = shop[idx]
        shop_sum[s] = shop_sum.get(s, 0) + v
    total += v * (j - i)
    i = j
sys.stdout.write("\n".join(map(str, ans)) + "\n")
```
