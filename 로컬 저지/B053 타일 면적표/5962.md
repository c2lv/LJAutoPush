### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
n, k = int(data[0]), int(data[1])


def count(x):
    total = 0
    full = min(n, x // n)
    total += full * n
    for i in range(full + 1, min(n, x) + 1):
        total += x // i
    return total


lo, hi = 1, min(n * n, k)
while lo < hi:
    mid = (lo + hi) // 2
    if count(mid) >= k:
        hi = mid
    else:
        lo = mid + 1
print(lo)
```
