### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
L, W, H = int(data[0]), int(data[1]), int(data[2])
n = int(data[3])
cnt = [0] * 20
for k in range(n):
    cnt[int(data[4 + 2 * k])] = int(data[5 + 2 * k])
filled = 0
ans = 0
for i in range(19, -1, -1):
    filled *= 8
    cap = (L >> i) * (W >> i) * (H >> i) - filled
    use = min(cnt[i], cap)
    filled += use
    ans += use
print(ans if filled == L * W * H else -1)
```
