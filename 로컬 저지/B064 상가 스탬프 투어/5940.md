### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
r, c = int(data[0]), int(data[1])
grid = data[2:2 + r]
bits = [1 << (ord(ch) - 65) for row in grid for ch in row]
nb = []
for i in range(r):
    for j in range(c):
        lst = []
        if i > 0:
            lst.append((i - 1) * c + j)
        if i < r - 1:
            lst.append((i + 1) * c + j)
        if j > 0:
            lst.append(i * c + j - 1)
        if j < c - 1:
            lst.append(i * c + j + 1)
        nb.append(lst)
best = 1
seen = set()
stack = [(0, bits[0], 1)]
while stack:
    p, mask, d = stack.pop()
    if d > best:
        best = d
        if best == 26:
            break
    for q in nb[p]:
        b = bits[q]
        if not mask & b:
            nm = mask | b
            key = (q, nm)
            if key not in seen:
                seen.add(key)
                stack.append((q, nm, d + 1))
print(best)
```
