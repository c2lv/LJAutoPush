### 파이썬 코드

```python
def solution(N, K, flowers, operations):
    cap = [row[:] for row in flowers]
    water = [[0] * N for _ in range(N)]
    idx = 0
    for _ in range(K):
        op = operations[idx][0]
        idx += 1
        if op == 1:
            for i in range(N):
                row = operations[idx]
                idx += 1
                ci = cap[i]
                wi = water[i]
                for j in range(N):
                    w = row[j]
                    if ci[j] > 0:
                        wi[j] += w
                        if wi[j] > ci[j]:
                            ci[j] = 0
                            wi[j] = 0
                    else:
                        wi[j] += w
        else:
            for i in range(N):
                ci = cap[i]
                wi = water[i]
                for j in range(N):
                    if ci[j] == 0 and wi[j] >= 10:
                        ci[j] = wi[j] // 2
                        wi[j] = 0
    return [[1 if c > 0 else 0 for c in row] for row in cap]
```
