### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
n = int(data[0])
rows = data[1:1 + n]

def count(grid):
    seen = [[False] * n for _ in range(n)]
    c = 0
    for i in range(n):
        for j in range(n):
            if not seen[i][j]:
                c += 1
                ch = grid[i][j]
                seen[i][j] = True
                stack = [(i, j)]
                while stack:
                    x, y = stack.pop()
                    for nx, ny in ((x + 1, y), (x - 1, y), (x, y + 1), (x, y - 1)):
                        if 0 <= nx < n and 0 <= ny < n and not seen[nx][ny] and grid[nx][ny] == ch:
                            seen[nx][ny] = True
                            stack.append((nx, ny))
    return c

print(count(rows), count([r.replace("R", "G") for r in rows]))
```
