### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
t = int(data[0])
out = []


def is_pal(s, i, j):
    sub = s[i:j + 1]
    return sub == sub[::-1]


for w in data[1:1 + t]:
    i, j = 0, len(w) - 1
    while i < j and w[i] == w[j]:
        i += 1
        j -= 1
    if i >= j:
        out.append("0")
    elif is_pal(w, i + 1, j) or is_pal(w, i, j - 1):
        out.append("1")
    else:
        out.append("2")
print("\n".join(out))
```
