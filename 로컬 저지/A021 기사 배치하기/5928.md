```python
def solution(N, K, board):
    knights = [(i, j) for i in range(N) for j in range(N) if board[i][j] == 0]
    jumps = ((-2, -1), (-2, 1), (-1, -2), (-1, 2), (1, -2), (1, 2), (2, -1), (2, 1))
    opts = []
    for x, y in knights:
        o = [(0, x * N + y)]
        for dx, dy in jumps:
            nx, ny = x + dx, y + dy
            if 0 <= nx < N and 0 <= ny < N:
                o.append((board[nx][ny], nx * N + ny))
        o.sort(reverse=True)
        opts.append(o)
    k = len(knights)
    suffix = [0] * (k + 1)
    for i in range(k - 1, -1, -1):
        suffix[i] = suffix[i + 1] + opts[i][0][0]
    used = set()
    best = 0

    def dfs(i, score):
        nonlocal best
        if score + suffix[i] <= best:
            return
        if i == k:
            best = score
            return
        for v, p in opts[i]:
            if p in used:
                continue
            used.add(p)
            dfs(i + 1, score + v)
            used.discard(p)

    dfs(0, 0)
    return best
```
