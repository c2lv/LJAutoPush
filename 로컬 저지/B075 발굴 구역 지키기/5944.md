### 파이썬 코드

```python
import sys
from itertools import combinations

data = sys.stdin.read().split()
n, m = int(data[0]), int(data[1])
g = [int(v) for v in data[2:2 + n * m]]
empties = [i for i in range(n * m) if g[i] == 0]
sources = [i for i in range(n * m) if g[i] == 2]
nb = []
for p in range(n * m):
    r, c = divmod(p, m)
    lst = []
    if r > 0:
        lst.append(p - m)
    if r < n - 1:
        lst.append(p + m)
    if c > 0:
        lst.append(p - 1)
    if c < m - 1:
        lst.append(p + 1)
    nb.append(lst)
total_empty = len(empties)
best = 0
for a, b, c in combinations(empties, 3):
    g[a] = g[b] = g[c] = 1
    seen = bytearray(n * m)
    stack = list(sources)
    for s in sources:
        seen[s] = 1
    infected = 0
    while stack:
        p = stack.pop()
        for q in nb[p]:
            if not seen[q] and g[q] == 0:
                seen[q] = 1
                infected += 1
                stack.append(q)
    safe = total_empty - 3 - infected
    if safe > best:
        best = safe
    g[a] = g[b] = g[c] = 0
print(best)
```
