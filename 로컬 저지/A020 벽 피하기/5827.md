### 파이썬 코드

```python
def solution(N, M, board):
    start = board[N - 1].index(2)
    reach = [False] * M
    reach[start] = True
    answer = 0
    for t in range(1, N):
        cur = board[N - t]
        nxt = board[N - 1 - t]
        new = [False] * M
        alive = False
        for c in range(M):
            if cur[c] == 1 or nxt[c] == 1:
                continue
            if reach[c] or (c > 0 and reach[c - 1]) or (c + 1 < M and reach[c + 1]):
                new[c] = True
                alive = True
        if not alive:
            break
        answer = t
        reach = new
    return answer
```
