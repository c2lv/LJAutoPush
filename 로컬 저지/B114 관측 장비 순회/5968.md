### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
n = int(data[0])
marks = data[1:1 + n]
h = [int(v) for v in data[1 + n:1 + n + n * n]]
W = n + 2
vals = sorted(set(h))
rank = {v: i for i, v in enumerate(vals)}
U = len(vals)
by_rank = [0] * U
pbit = 0
kmask = 0
req = []
for r in range(n):
    for c in range(n):
        b = 1 << ((r + 1) * W + c + 1)
        hv = h[r * n + c]
        by_rank[rank[hv]] |= b
        ch = marks[r][c]
        if ch == "P":
            pbit = b
            req.append(hv)
        elif ch == "K":
            kmask |= b
            req.append(hv)
le = []
acc = 0
for m in by_rank:
    acc |= m
    le.append(acc)
need = pbit | kmask


def feasible(i, j):
    allowed = le[j] & ~(le[i - 1] if i > 0 else 0)
    if allowed & need != need:
        return False
    reach = pbit
    while True:
        x = reach | (reach << 1) | (reach >> 1)
        nr = (x | (x << W) | (x >> W)) & allowed
        if nr == reach:
            break
        reach = nr
        if reach & kmask == kmask:
            return True
    return reach & kmask == kmask


lo_lim = rank[min(req)]
j = rank[max(req)]
best = None
for i in range(lo_lim + 1):
    if j < i:
        j = i
    while j < U and not feasible(i, j):
        j += 1
    if j == U:
        break
    d = vals[j] - vals[i]
    if best is None or d < best:
        best = d
print(best)
```
