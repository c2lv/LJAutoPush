### 파이썬 코드

```python
import sys
from collections import deque

data = sys.stdin.read().split()
t = int(data[0])
pos = 1
moves = ((-2, -1), (-2, 1), (-1, -2), (-1, 2), (1, -2), (1, 2), (2, -1), (2, 1))
out = []
for _ in range(t):
    l = int(data[pos])
    sx, sy, tx, ty = (int(v) for v in data[pos + 1:pos + 5])
    pos += 5
    dist = [[-1] * l for _ in range(l)]
    dist[sx][sy] = 0
    q = deque([(sx, sy)])
    while q:
        x, y = q.popleft()
        if x == tx and y == ty:
            break
        d = dist[x][y] + 1
        for dx, dy in moves:
            nx, ny = x + dx, y + dy
            if 0 <= nx < l and 0 <= ny < l and dist[nx][ny] == -1:
                dist[nx][ny] = d
                q.append((nx, ny))
    out.append(str(dist[tx][ty]))
print("\n".join(out))
```
