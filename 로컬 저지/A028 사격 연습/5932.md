### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
n = int(data[0])
k = int(data[1])
pos = 2
init = []
for i in range(n):
    init.append([int(v) for v in data[pos:pos + n]])
    pos += n
atk = [int(v) for v in data[pos:pos + k]]
cur = [row[:] for row in init]
best = 0


def dfs(t, score):
    global best
    if t == k:
        if score > best:
            best = score
        return
    a = atk[t]
    for r in range(n):
        c = 0
        while c < n and init[r][c] == 0:
            c += 1
        if c == n:
            dfs(t + 1, score)
            continue
        iv = init[r][c]
        cv = cur[r][c]
        if iv >= 10:
            init[r][c] = 0
            cur[r][c] = 0
            dfs(t + 1, score + iv)
            init[r][c] = iv
            cur[r][c] = cv
            continue
        nv = cv - a
        if nv > 0:
            cur[r][c] = nv
            dfs(t + 1, score)
            cur[r][c] = cv
            continue
        init[r][c] = 0
        cur[r][c] = 0
        child = iv // 4
        spawned = []
        if child > 0:
            for nr, nc in ((r + 1, c), (r - 1, c), (r, c + 1), (r, c - 1)):
                if 0 <= nr < n and 0 <= nc < n and init[nr][nc] == 0:
                    init[nr][nc] = child
                    cur[nr][nc] = child
                    spawned.append((nr, nc))
        dfs(t + 1, score + iv)
        for nr, nc in spawned:
            init[nr][nc] = 0
            cur[nr][nc] = 0
        init[r][c] = iv
        cur[r][c] = cv


dfs(0, 0)
print(best)
```
