### 파이썬 코드

```python
def solution(N, arr, conditions):
    def ok(c, left, right):
        if c == 1:
            return left > right
        if c == 2:
            return left < right
        if c == 3:
            return left == right
        return True

    full = 1 << N
    dp = [[0] * N for _ in range(full)]
    for i in range(N):
        dp[1 << i][i] = 1
    for mask in range(1, full):
        k = bin(mask).count("1")
        if k == N:
            continue
        c = conditions[k - 1]
        row = dp[mask]
        for last in range(N):
            v = row[last]
            if v == 0:
                continue
            for nx in range(N):
                if not (mask >> nx) & 1 and ok(c, arr[last], arr[nx]):
                    dp[mask | (1 << nx)][nx] += v
    return sum(dp[full - 1])
```
