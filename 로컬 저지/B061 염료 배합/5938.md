### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
n = int(data[0])
items = [(int(data[1 + 2 * i]), int(data[2 + 2 * i])) for i in range(n)]
best = None
for mask in range(1, 1 << n):
    p = 1
    s = 0
    for i in range(n):
        if (mask >> i) & 1:
            p *= items[i][0]
            s += items[i][1]
    d = abs(p - s)
    if best is None or d < best:
        best = d
print(best)
```
