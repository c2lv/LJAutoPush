### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
n = int(data[0])
to = [0] + [int(v) for v in data[1:1 + n]]
indeg = [0] * (n + 1)
for i in range(1, n + 1):
    indeg[to[i]] += 1
alive = [True] * (n + 1)
stack = [i for i in range(1, n + 1) if indeg[i] == 0]
while stack:
    u = stack.pop()
    alive[u] = False
    v = to[u]
    indeg[v] -= 1
    if indeg[v] == 0:
        stack.append(v)
res = [i for i in range(1, n + 1) if alive[i]]
print(len(res))
for v in res:
    print(v)
```
