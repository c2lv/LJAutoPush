### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
n = int(data[0])
g = [list(data[1 + i]) for i in range(n)]
sizes = []
for i in range(n):
    for j in range(n):
        if g[i][j] == "1":
            g[i][j] = "0"
            stack = [(i, j)]
            c = 0
            while stack:
                x, y = stack.pop()
                c += 1
                for nx, ny in ((x + 1, y), (x - 1, y), (x, y + 1), (x, y - 1)):
                    if 0 <= nx < n and 0 <= ny < n and g[nx][ny] == "1":
                        g[nx][ny] = "0"
                        stack.append((nx, ny))
            sizes.append(c)
sizes.sort()
print(len(sizes))
print("\n".join(map(str, sizes)))
```
