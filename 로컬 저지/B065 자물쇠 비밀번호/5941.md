```python
import sys

data = sys.stdin.read().split()
k = int(data[0])
signs = data[1:1 + k]


def search(order):
    used = [False] * 10
    path = []

    def dfs(i):
        if i == k + 1:
            return True
        for d in order:
            if used[d]:
                continue
            if i > 0:
                prev = path[-1]
                if signs[i - 1] == "<" and not prev < d:
                    continue
                if signs[i - 1] == ">" and not prev > d:
                    continue
            used[d] = True
            path.append(d)
            if dfs(i + 1):
                return True
            path.pop()
            used[d] = False
        return False

    dfs(0)
    return "".join(map(str, path))


print(search(range(9, -1, -1)))
print(search(range(10)))
```
