### 파이썬 코드

```python
import sys
from collections import deque

s, t = map(int, sys.stdin.read().split()[:2])
LIM = 1000000000
if s == t:
    print(0)
else:
    seen = {s: ""}
    q = deque([s])
    ans = -1
    while q:
        x = q.popleft()
        path = seen[x]
        if x == t:
            ans = path
            break
        cands = [(x * x, "*"), (x + x, "+"), (0, "-")]
        if x != 0:
            cands.append((1, "/"))
        for y, op in cands:
            if y > LIM or y in seen:
                continue
            seen[y] = path + op
            q.append(y)
    print(ans)
```
