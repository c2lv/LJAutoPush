```python
import heapq

def solution(N, arr):
    h = [v for row in arr for v in row]
    total = N * N
    INF = float("inf")
    dist = [INF] * total
    dist[0] = 0
    pq = [(0, 0)]
    target = total - 1
    while pq:
        d, p = heapq.heappop(pq)
        if d > dist[p]:
            continue
        if p == target:
            return d
        x, y = divmod(p, N)
        hp = h[p]
        if x > 0:
            q = p - N
            nd = d + abs(hp - h[q])
            if nd < dist[q]:
                dist[q] = nd
                heapq.heappush(pq, (nd, q))
        if x < N - 1:
            q = p + N
            nd = d + abs(hp - h[q])
            if nd < dist[q]:
                dist[q] = nd
                heapq.heappush(pq, (nd, q))
        if y > 0:
            q = p - 1
            nd = d + abs(hp - h[q])
            if nd < dist[q]:
                dist[q] = nd
                heapq.heappush(pq, (nd, q))
        if y < N - 1:
            q = p + 1
            nd = d + abs(hp - h[q])
            if nd < dist[q]:
                dist[q] = nd
                heapq.heappush(pq, (nd, q))
    return dist[target]
```
