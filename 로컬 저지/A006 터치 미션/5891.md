```python
def solution(N, board):
    W = N + 2
    size = W * W
    wall = [True] * size
    targets = []
    start = 0
    for i in range(N):
        row = board[i]
        base = (i + 1) * W + 1
        for j in range(N):
            ch = row[j]
            p = base + j
            if ch != "X":
                wall[p] = False
            if ch == "O":
                targets.append(p)
            elif ch == "P":
                start = p
    dirs = (1, -1, W, -W)

    def bfs():
        dist = [-1] * size
        dist[start] = 0
        cur = [start]
        d = 0
        while cur:
            d += 1
            nxt = []
            for p in cur:
                for dd in dirs:
                    q = p + dd
                    if dist[q] == -1 and not wall[q]:
                        dist[q] = d
                        nxt.append(q)
            cur = nxt
        return dist

    INF = float("inf")
    best = INF
    d0 = bfs()
    ds = sorted(d0[t] for t in targets if d0[t] != -1)
    if ds:
        best = min(best, ds[-1])
        if len(ds) >= 2:
            best = min(best, ds[-2])
    for i in range(N):
        base = (i + 1) * W + 1
        for j in range(N):
            p = base + j
            if board[i][j] != "X":
                continue
            if d0[p + 1] == -1 and d0[p - 1] == -1 and d0[p + W] == -1 and d0[p - W] == -1:
                continue
            wall[p] = False
            d1 = bfs()
            wall[p] = True
            far = -1
            for t in targets:
                if d1[t] > far:
                    far = d1[t]
            if far != -1 and far < best:
                best = far
    return -1 if best == INF else best
```
