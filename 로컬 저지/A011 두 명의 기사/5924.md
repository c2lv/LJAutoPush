```python
def solution(N, board):
    knights = []
    hero = None
    enemies = []
    for i in range(N):
        for j in range(N):
            ch = board[i][j]
            if ch == "K":
                knights.append((i, j))
            elif ch == "H":
                hero = (i, j)
            elif ch == "E":
                enemies.append((i, j))
    jumps = ((-2, -1), (-2, 1), (-1, -2), (-1, 2), (1, -2), (1, 2), (2, -1), (2, 1))

    def options(k, other):
        res = [k]
        for dx, dy in jumps:
            x, y = k[0] + dx, k[1] + dy
            if 0 <= x < N and 0 <= y < N:
                ch = board[x][y]
                if ch in ("B", "E") or (x, y) == other:
                    res.append((x, y))
        return res

    k1, k2 = knights
    opt1 = options(k1, k2)
    opt2 = options(k2, k1)
    enemy_set = set(enemies)
    best = -1
    seen = set()
    for p1 in opt1:
        for p2 in opt2:
            if p1 == p2:
                continue
            key = (min(p1, p2), max(p1, p2))
            if key in seen:
                continue
            seen.add(key)
            block = [[ch == "X" for ch in row] for row in board]
            block[p1[0]][p1[1]] = True
            block[p2[0]][p2[1]] = True
            alive = enemy_set - {p1, p2}
            if not alive:
                return 0
            dist = [[-1] * N for _ in range(N)]
            dist[hero[0]][hero[1]] = 0
            cur = [hero]
            found = 0
            d = 0
            while cur and not found:
                d += 1
                nxt = []
                for x, y in cur:
                    for nx, ny in ((x + 1, y), (x - 1, y), (x, y + 1), (x, y - 1)):
                        if 0 <= nx < N and 0 <= ny < N and dist[nx][ny] == -1 and not block[nx][ny]:
                            dist[nx][ny] = d
                            if (nx, ny) in alive:
                                found = d
                            nxt.append((nx, ny))
                cur = nxt
            if found == 0:
                return 0
            best = max(best, found)
    return best
```
