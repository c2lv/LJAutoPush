### 파이썬 코드

```python
import sys

data = sys.stdin.buffer.read().split()
n, k = int(data[0]), int(data[1])
odd = [-1]
i = 0
for tok in data[2:2 + n]:
    if tok[-1] in b"13579":
        odd.append(i)
    i += 1
odd.append(n)
cnt = len(odd) - 2
if cnt <= k:
    print(n - cnt)
else:
    best = 0
    for j in range(1, cnt - k + 2):
        v = odd[j + k] - odd[j - 1] - 1 - k
        if v > best:
            best = v
    print(best)
```
