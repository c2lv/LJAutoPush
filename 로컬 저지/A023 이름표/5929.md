```python
from itertools import accumulate

def solution(N, M, students, queries):
    L = 300000
    diff = [0] * (L + 2)
    for t, k in students:
        e = t + k
        if e > L + 1:
            e = L + 1
        diff[t] += 1
        diff[e] -= 1
    cnt = list(accumulate(diff))
    pre = list(accumulate(cnt))
    return [pre[e] - pre[s - 1] for s, e in queries]
```
