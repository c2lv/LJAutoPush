### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
n = int(data[0])
pos = 1
dp = []
for i in range(1, n + 1):
    row = [int(x) for x in data[pos:pos + i]]
    pos += i
    for j in range(i):
        best = 0
        if j < i - 1:
            best = dp[j]
        if j > 0 and dp[j - 1] > best:
            best = dp[j - 1]
        row[j] += best
    dp = row
print(max(dp))
```
