### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
n, m = int(data[0]), int(data[1])
g = [int(v) for v in data[2:2 + n * m]]
nb = []
for p in range(n * m):
    r, c = divmod(p, m)
    lst = []
    if r > 0:
        lst.append(p - m)
    if r < n - 1:
        lst.append(p + m)
    if c > 0:
        lst.append(p - 1)
    if c < m - 1:
        lst.append(p + 1)
    nb.append(lst)
land = sum(g)
years = 0
while land:
    outside = bytearray(n * m)
    outside[0] = 1
    stack = [0]
    while stack:
        p = stack.pop()
        for q in nb[p]:
            if not outside[q] and g[q] == 0:
                outside[q] = 1
                stack.append(q)
    melt = []
    for p in range(n * m):
        if g[p] == 1:
            k = 0
            for q in nb[p]:
                if outside[q]:
                    k += 1
            if k >= 2:
                melt.append(p)
    for p in melt:
        g[p] = 0
    land -= len(melt)
    years += 1
print(years)
```
