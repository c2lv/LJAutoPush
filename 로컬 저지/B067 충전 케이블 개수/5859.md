### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
t = int(data[0])
pos = 1
out = []
for _ in range(t):
    m, n, k = int(data[pos]), int(data[pos + 1]), int(data[pos + 2])
    pos += 3
    cells = set()
    for i in range(k):
        cells.add((int(data[pos]), int(data[pos + 1])))
        pos += 2
    cnt = 0
    while cells:
        cnt += 1
        stack = [cells.pop()]
        while stack:
            x, y = stack.pop()
            for nb in ((x + 1, y), (x - 1, y), (x, y + 1), (x, y - 1)):
                if nb in cells:
                    cells.remove(nb)
                    stack.append(nb)
    out.append(str(cnt))
print("\n".join(out))
```
