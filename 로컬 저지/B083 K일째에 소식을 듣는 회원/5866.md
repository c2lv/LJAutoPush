### 파이썬 코드

```python
import sys

data = sys.stdin.buffer.read().split()
n, m, k, x = int(data[0]), int(data[1]), int(data[2]), int(data[3])
adj = [[] for _ in range(n + 1)]
for i in range(m):
    adj[int(data[4 + 2 * i])].append(int(data[5 + 2 * i]))
seen = [False] * (n + 1)
seen[x] = True
cur = [x]
d = 0
while cur and d < k:
    nxt = []
    for u in cur:
        for v in adj[u]:
            if not seen[v]:
                seen[v] = True
                nxt.append(v)
    cur = nxt
    d += 1
if d == k and cur:
    cur.sort()
    print("\n".join(map(str, cur)))
else:
    print(-1)
```
