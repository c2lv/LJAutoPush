### 파이썬 코드

```python
import sys

data = sys.stdin.read().split()
n = int(data[0])
nums = [int(v) for v in data[1:1 + n]]
ops = [int(v) for v in data[1 + n:5 + n]]
hi = None
lo = None


def dfs(i, val):
    global hi, lo
    if i == n:
        if hi is None or val > hi:
            hi = val
        if lo is None or val < lo:
            lo = val
        return
    x = nums[i]
    for k in range(4):
        if ops[k] == 0:
            continue
        ops[k] -= 1
        if k == 0:
            nv = val + x
        elif k == 1:
            nv = val - x
        elif k == 2:
            nv = val * x
        else:
            nv = -((-val) // x) if val < 0 else val // x
        dfs(i + 1, nv)
        ops[k] += 1


dfs(1, nums[0])
print(hi)
print(lo)
```
