```python
import sys
from itertools import combinations_with_replacement

n, m = map(int, sys.stdin.read().split()[:2])
out = ["".join("%d " % x for x in p) for p in combinations_with_replacement(range(1, n + 1), m)]
sys.stdout.write("\n".join(out) + "\n\n")
```
