### 파이썬 코드

```python
import sys

lines = sys.stdin.read().split("\n")
n, m, h = map(int, lines[0].split()[:3])
dp = [0] * (h + 1)
dp[0] = 1
for i in range(1, n + 1):
    boards = [int(v) for v in lines[i].split()]
    nd = dp[:]
    for b in boards:
        for s in range(h - b + 1):
            if dp[s]:
                nd[s + b] = (nd[s + b] + dp[s]) % 10007
    dp = nd
print(dp[h] % 10007)
```
